package com.nutriguard.api;

import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;

@Entity
public class FoodProduct {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    private String name;
    private String category;
    private double sugar;
    private double sodium;
    private double fat;
    private String riskLevel;

    public FoodProduct() { }

    public FoodProduct(String name, String category, double sugar, double sodium, double fat, String riskLevel) {
        this.name = name;
        this.category = category;
        this.sugar = sugar;
        this.sodium = sodium;
        this.fat = fat;
        this.riskLevel = riskLevel;
    }

    public Long getId() { return id; }
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public String getCategory() { return category; }
    public void setCategory(String category) { this.category = category; }
    public double getSugar() { return sugar; }
    public void setSugar(double sugar) { this.sugar = sugar; }
    public double getSodium() { return sodium; }
    public void setSodium(double sodium) { this.sodium = sodium; }
    public double getFat() { return fat; }
    public void setFat(double fat) { this.fat = fat; }
    public String getRiskLevel() { return riskLevel; }
    public void setRiskLevel(String riskLevel) { this.riskLevel = riskLevel; }
}

package com.nutriguard.api;

import java.util.List;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/foods")
@CrossOrigin
public class FoodProductController {
    private final FoodProductRepository repository;

    public FoodProductController(FoodProductRepository repository) {
        this.repository = repository;
    }

    @GetMapping
    public List<FoodProduct> getAll() { return repository.findAll(); }

    @GetMapping("/{id}")
    public ResponseEntity<FoodProduct> getOne(@PathVariable Long id) {
        return repository.findById(id).map(ResponseEntity::ok).orElse(ResponseEntity.notFound().build());
    }

    @PostMapping
    public ResponseEntity<FoodProduct> create(@RequestBody FoodProduct food) {
        food.setRiskLevel(calculateRisk(food));
        return ResponseEntity.status(201).body(repository.save(food));
    }

    @PutMapping("/{id}")
    public ResponseEntity<FoodProduct> update(@PathVariable Long id, @RequestBody FoodProduct input) {
        return repository.findById(id).map(food -> {
            food.setName(input.getName());
            food.setCategory(input.getCategory());
            food.setSugar(input.getSugar());
            food.setSodium(input.getSodium());
            food.setFat(input.getFat());
            food.setRiskLevel(calculateRisk(food));
            return ResponseEntity.ok(repository.save(food));
        }).orElse(ResponseEntity.notFound().build());
    }

    @PatchMapping("/{id}")
    public ResponseEntity<FoodProduct> patchRisk(@PathVariable Long id, @RequestBody FoodProduct input) {
        return repository.findById(id).map(food -> {
            food.setRiskLevel(input.getRiskLevel());
            return ResponseEntity.ok(repository.save(food));
        }).orElse(ResponseEntity.notFound().build());
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable Long id) {
        if (!repository.existsById(id)) return ResponseEntity.notFound().build();
        repository.deleteById(id);
        return ResponseEntity.noContent().build();
    }

    private String calculateRisk(FoodProduct food) {
        if (food.getSugar() > 12 || food.getSodium() > 450 || food.getFat() > 20) return "High Risk";
        if (food.getSugar() > 6 || food.getSodium() > 300 || food.getFat() > 12) return "Moderate Risk";
        return "Low Risk";
    }
}

`$lang
from datetime import datetime, timedelta, timezone

import jwt
from flask import Flask, jsonify, request
from werkzeug.security import check_password_hash, generate_password_hash

app = Flask(__name__)
app.config['SECRET_KEY'] = 'college-project-secret-key'

# Simple in-memory store for the lab demonstration.
users = {}


@app.post('/register')
def register():
    data = request.get_json() or {}
    username = data.get('username', '').strip()
    password = data.get('password', '')

    if not username or not password:
        return jsonify({'message': 'Username and password are required'}), 400
    if username in users:
        return jsonify({'message': 'User already exists'}), 400

    users[username] = generate_password_hash(password)
    return jsonify({'message': 'User registered successfully'}), 201


@app.post('/login')
def login():
    data = request.get_json() or {}
    username = data.get('username', '').strip()
    password = data.get('password', '')

    if username not in users or not check_password_hash(users[username], password):
        return jsonify({'message': 'Invalid username or password'}), 401

    payload = {
        'username': username,
        'exp': datetime.now(timezone.utc) + timedelta(hours=1),
    }
    token = jwt.encode(payload, app.config['SECRET_KEY'], algorithm='HS256')
    return jsonify({'token': token})


@app.get('/dashboard')
def dashboard():
    header = request.headers.get('Authorization', '')
    if not header.startswith('Bearer '):
        return jsonify({'message': 'Bearer token is required'}), 401

    token = header.split(' ', 1)[1]
    try:
        user = jwt.decode(token, app.config['SECRET_KEY'], algorithms=['HS256'])
        return jsonify({'message': f"Welcome {user['username']}", 'access': 'granted'})
    except jwt.ExpiredSignatureError:
        return jsonify({'message': 'Token has expired'}), 401
    except jwt.InvalidTokenError:
        return jsonify({'message': 'Invalid token'}), 401


if __name__ == '__main__':
    app.run(debug=True, port=5000)

import React, { createContext, useContext, useState } from 'react';

const NutriContext = createContext();

export const NutriProvider = ({ children }) => {
  const [foodRecords, setFoodRecords] = useState([]);
  const [activeFood, setActiveFood] = useState(null);

  const addFoodRecord = (newRecord) => {
    setFoodRecords((prevRecords) => [...prevRecords, newRecord]);
    setActiveFood(newRecord);
  };

  const updateFoodRecord = (updatedRecord) => {
    setFoodRecords((previous) => previous.map((record) => record.id === updatedRecord.id ? updatedRecord : record));
    setActiveFood(updatedRecord);
  };

  const deleteFoodRecord = (id) => {
    setFoodRecords((previous) => previous.filter((record) => record.id !== id));
    setActiveFood(null);
  };

  return (
    <NutriContext.Provider value={{ foodRecords, activeFood, addFoodRecord, updateFoodRecord, deleteFoodRecord }}>
      {children}
    </NutriContext.Provider>
  );
};

export const useNutriContext = () => useContext(NutriContext);

`$lang
class FoodProfile {
  constructor({ id, name, category, sugar, sodium, fat, price, rating }) {
    this.id = id;
    this.name = name;
    this.category = category;
    this.sugar = Number(sugar || 0);
    this.sodium = Number(sodium || 0);
    this.fat = Number(fat || 0);
    this.price = Number(price || 0);
    this.rating = Number(rating || 0);
  }

  calculateRisk() {
    if (this.sugar > 12 || this.sodium > 450 || this.fat > 20) return 'High Risk';
    if (this.sugar > 6 || this.sodium > 300) return 'Moderate Risk';
    return 'Low Risk';
  }

  getDetails() {
    return `[${this.id}] ${this.name} | Category: ${this.category} | Risk: ${this.calculateRisk()}`;
  }
}

class BeverageProfile extends FoodProfile {
  calculateRisk() {
    if (this.sugar > 10 || this.sodium > 200 || this.fat > 5) return 'High Risk';
    if (this.sugar > 6 || this.sodium > 120) return 'Moderate Risk';
    return 'Low Risk';
  }
}

class SnackProfile extends FoodProfile {
  calculateRisk() {
    if (this.sodium > 700 || this.fat > 18 || this.sugar > 10) return 'High Risk';
    if (this.sodium > 300 || this.fat > 12) return 'Moderate Risk';
    return 'Low Risk';
  }
}

class BakeryProfile extends FoodProfile {
  calculateRisk() {
    if (this.sugar > 14 || this.sodium > 350 || this.fat > 22) return 'High Risk';
    if (this.sugar > 8 || this.sodium > 250) return 'Moderate Risk';
    return 'Low Risk';
  }
}

export const createFoodProfile = (food) => {
  switch (food.category) {
    case 'Beverage':
      return new BeverageProfile(food);
    case 'Snack':
      return new SnackProfile(food);
    case 'Bakery':
      return new BakeryProfile(food);
    default:
      return new FoodProfile(food);
  }
};

export const buildFoodRegistry = (foods = []) => {
  const registryList = foods.map((food) => createFoodProfile(food));
  const registryMap = Object.fromEntries(registryList.map((item) => [item.id, item]));
  const uniqueCategories = [...new Set(registryList.map((item) => item.category))];
  const uniqueSkills = new Set(registryList.flatMap((item) => [item.category]));

  return {
    registryList,
    registryMap,
    uniqueCategories,
    uniqueSkills,
    totalRiskRevenue: registryList.reduce((sum, item) => sum + Number(item.price || 0), 0)
  };
};

export const classifyRisk = (food) => {
  const sugar = Number(food.sugar || 0);
  const sodium = Number(food.sodium || 0);
  const fat = Number(food.fat || 0);
  const additives = food.additives || 'Low';
  if (sugar > 12 || sodium > 450 || fat > 20 || additives === 'High') return { level: 'High Risk', reason: 'One or more nutrient values are high.' };
  if (sugar > 6 || sodium > 300 || fat > 12 || additives === 'Medium') return { level: 'Moderate Risk', reason: 'Some nutrient values need monitoring.' };
  return { level: 'Low Risk', reason: 'Nutrient values are within the basic limits.' };
};
